# Packages this setup needs — Debian 13 "trixie" and later.
# Install:  grep -v '^#' packages.debian.txt | xargs sudo apt install
#
# The Arch list is packages.txt — same setup, Arch names. install.sh detects the
# host and points at the right one. Components contrib/non-free/non-free-firmware
# must be enabled (trixie's installer offers non-free-firmware; the GPU firmware
# and some fonts live there).
#
# Names that differ from the Arch list:
#   xorg-xrandr / xorg-xsetroot -> x11-xserver-utils
#   xorg-xinput                -> xinput
#   setxkbmap + xkbcomp        -> x11-xkb-utils    (used by ~/.xprofile)
#   xprop & friends            -> x11-utils
#   polkit-gnome               -> mate-polkit
#   networkmanager             -> network-manager
#   bluez-utils                -> bluez
#   python                     -> python3
#   mesa                       -> libgl1-mesa-dri + mesa-utils
#   noto-fonts                 -> fonts-noto (+ fonts-noto-color-emoji)

# sudo is NOT part of "standard system utilities" — the Debian installer only
# installs it when you leave the root password EMPTY. Listed so a box that set a
# root password still ends up with it. Install it as root BEFORE running the
# "apt install everything" line:  apt install sudo && usermod -aG sudo <you>
# (then log out/in). See README -> "Debian (apt) machines".
sudo

i3-wm
i3status
i3lock

# VSync for X11 — i3 is not a compositor, so without this windowed content tears.
picom

# ⚠️ Terminus is the desktop font, and the setup asks for the NERD-PATCHED family
# "Terminess Nerd Font Mono" (terminal, bar, rofi, GTK, Thunar). Debian only has
# the PLAIN fonts-terminus / fonts-firacode, which do NOT provide the Nerd
# families, so install the nerd builds instead of the Debian packages:
#   https://github.com/ryanoasis/nerd-fonts/releases  ->  Terminus.zip, FiraCode.zip
#   unzip -o Terminus.zip FiraCode.zip -d ~/.local/share/fonts && fc-cache -f
# local-bin/terminus-bold then derives the Bold-only family the browser asks for
# (needs uv — see below).

# uv — used by local-bin/terminus-bold to run fontTools in a throwaway env.
# Not packaged in trixie:  curl -LsSf https://astral.sh/uv/install.sh | sh
# (or: cargo install uv)

# Terminal niceties the repo ships configs for
starship

xdg-user-dirs
